  kubectl --namespace {{ .Release.Namespace }} port-forward svc/{{ include "observaki.fullname" . }}-api 8080:{{ .Values.api.service.port }}

  kubectl --namespace {{ .Release.Namespace }} port-forward svc/{{ include "observaki.fullname" . }}-gateway 4318:{{ .Values.gateway.service.httpPort }}
  OBSERVAKI_OTLP_HTTP_URL=http://127.0.0.1:4318 ./deploy/compose/sample.sh
